CREATE TYPE "staff_role" AS ENUM (
  'ADMIN',
  'RECEPTION'
);

CREATE TYPE "client_status" AS ENUM (
  'ACTIVE',
  'INACTIVE',
  'SUSPENDED'
);

CREATE TYPE "skill_level" AS ENUM (
  'BEGINNER',
  'INTERMEDIATE',
  'ADVANCED',
  'PRO'
);

CREATE TYPE "client_skill_category" AS ENUM (
  'SWIMMING_TECHNIQUE',
  'WATER_SAFETY',
  'ENDURANCE'
);

CREATE TYPE "instructor_skill_category" AS ENUM (
  'CERTIFICATION',
  'SPECIALIZATION',
  'LIFESAVING'
);

CREATE TYPE "lesson_type" AS ENUM (
  'INDIVIDUAL',
  'SEMI_INDIVIDUAL',
  'GROUP',
  'AQUAAEROBICS',
  'SPECIAL_EVENT'
);

CREATE TYPE "pass_status" AS ENUM (
  'ACTIVE',
  'FROZEN',
  'EXPIRED',
  'EXHAUSTED'
);

CREATE TYPE "booking_status" AS ENUM (
  'BOOKED',
  'ATTENDED',
  'CANCELLED_ON_TIME',
  'NO_SHOW',
  'RESCHEDULED'
);

CREATE TYPE "membership_status" AS ENUM (
  'WAITLISTED',
  'ACTIVE',
  'RESIGNED',
  'COMPLETED'
);

CREATE TABLE "users" (
  "id" uuid PRIMARY KEY,
  "email" varchar UNIQUE NOT NULL,
  "password_hash" varchar NOT NULL,
  "first_name" varchar NOT NULL,
  "last_name" varchar NOT NULL,
  "phone" varchar
);

CREATE TABLE "staff_members" (
  "id" uuid PRIMARY KEY,
  "staff_role" staff_role NOT NULL
);

CREATE TABLE "instructors" (
  "id" uuid PRIMARY KEY
);

CREATE TABLE "clients" (
  "id" uuid PRIMARY KEY,
  "first_name" varchar NOT NULL,
  "last_name" varchar NOT NULL,
  "email" varchar,
  "phone" varchar,
  "birth_date" date,
  "skill_level" skill_level,
  "status" client_status NOT NULL
);

CREATE TABLE "guardians" (
  "id" uuid PRIMARY KEY,
  "first_name" varchar NOT NULL,
  "last_name" varchar NOT NULL,
  "phone" varchar,
  "email" varchar
);

CREATE TABLE "client_guardians" (
  "client_id" uuid,
  "guardians_id" uuid,
  PRIMARY KEY ("client_id", "guardians_id")
);

CREATE TABLE "instructor_skills" (
  "id" uuid PRIMARY KEY,
  "code" varchar NOT NULL,
  "name" varchar NOT NULL,
  "category" instructor_skill_category NOT NULL,
  "description" text
);

CREATE TABLE "client_skills" (
  "id" uuid PRIMARY KEY,
  "code" varchar NOT NULL,
  "name" varchar NOT NULL,
  "category" client_skill_category NOT NULL,
  "description" text
);

CREATE TABLE "client_mastered_skills" (
  "client_id" uuid,
  "client_skill_id" uuid,
  PRIMARY KEY ("client_id", "client_skill_id")
);

CREATE TABLE "instructor_qualifications" (
  "id" uuid PRIMARY KEY,
  "instructor_id" uuid,
  "instructor_skill_id" uuid,
  "lesson_type" lesson_type NOT NULL,
  "skill_level" skill_level NOT NULL,
  "valid_from" date,
  "valid_until" date
);

CREATE TABLE "instructor_availabilities" (
  "id" uuid PRIMARY KEY,
  "instructor_id" uuid,
  "date" date NOT NULL,
  "start_time" time NOT NULL,
  "end_time" time NOT NULL,
  "is_available" boolean NOT NULL
);

CREATE TABLE "facilities" (
  "id" uuid PRIMARY KEY,
  "name" varchar NOT NULL,
  "address" varchar,
  "city" varchar
);

CREATE TABLE "swim_groups" (
  "id" uuid PRIMARY KEY,
  "name" varchar NOT NULL,
  "target_skill_level" skill_level,
  "min_age" int,
  "max_age" int,
  "max_participants" int
);

CREATE TABLE "instructor_swim_groups" (
  "instructor_id" uuid,
  "swim_group_id" uuid,
  PRIMARY KEY ("instructor_id", "swim_group_id")
);

CREATE TABLE "group_memberships" (
  "id" uuid PRIMARY KEY,
  "client_id" uuid,
  "swim_group_id" uuid,
  "joined_at" date NOT NULL,
  "left_at" date,
  "status" membership_status NOT NULL
);

CREATE TABLE "lesson_sessions" (
  "id" uuid PRIMARY KEY,
  "instructor_id" uuid,
  "facility_id" uuid,
  "swim_group_id" uuid,
  "instructor_availability_id" uuid,
  "start_time" timestamp NOT NULL,
  "duration_minutes" smallint NOT NULL,
  "type" lesson_type NOT NULL,
  "is_cancelled" boolean NOT NULL
);

CREATE TABLE "passes" (
  "id" uuid PRIMARY KEY,
  "client_id" uuid,
  "total_lessons" int NOT NULL,
  "used_lessons" int NOT NULL,
  "valid_from" date NOT NULL,
  "valid_until" date NOT NULL,
  "status" pass_status NOT NULL
);

CREATE TABLE "bookings" (
  "id" uuid PRIMARY KEY,
  "client_id" uuid,
  "lesson_session_id" uuid,
  "pass_id" uuid,
  "booked_at" timestamp NOT NULL,
  "status" booking_status NOT NULL
);

CREATE TABLE "contract_documents" (
  "id" uuid PRIMARY KEY,
  "client_id" uuid,
  "guardian_id" uuid,
  "title" varchar NOT NULL,
  "template_url" varchar,
  "signed_file_url" varchar,
  "is_signed" boolean NOT NULL,
  "generated_at" timestamp NOT NULL
);
